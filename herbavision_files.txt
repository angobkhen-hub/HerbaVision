from passlib.context import CryptContext
from jose import jwt

ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = int(os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", 60))

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)


def create_access_token(data: dict) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)


def decode_access_token(token: str) -> dict | None:
    try:
        return jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
    except Exception:
        return None

def get_current_user(
    token: str = Depends(oauth2_scheme),
    db: Session = Depends(get_db),
) -> User:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )

    payload = decode_access_token(token)
    if payload is None:
        raise credentials_exception

    user_id = payload.get("sub")
    if user_id is None:
        raise credentials_exception

    user = db.query(User).filter(User.id == int(user_id)).first()
    if user is None:
        raise credentials_exception

    return user

r"""
Build the "what do the six plants look like" statistics used to reject unknown leaves.
Run this ONCE (and again whenever you retrain leaf_model.pt).

    python build_leaf_stats.py
    python build_leaf_stats.py --data C:\Users\user\Downloads\dataset --unknown C:\leaf_tests\unknown

It reads the training photos (data\train\<class>\*.jpg), measures each plant's typical
features with the existing model, and saves leaf_stats.pt next to this file.
The threshold is set from the validation photos (data\val), so real plants should pass.
It does NOT change leaf_model.pt and needs no new photos and no internet.
"""

import argparse
import os
import sys

import torch
import torch.nn as nn
from PIL import Image
from torchvision import models, transforms

HERE = os.path.dirname(os.path.abspath(__file__))
MODEL_PATH = os.path.join(HERE, "leaf_model.pt")
STATS_PATH = os.path.join(HERE, "leaf_stats.pt")
DEFAULT_DATA = r"C:\Users\user\Downloads\dataset"
IMAGE_EXTS = (".jpg", ".jpeg", ".png", ".webp", ".bmp")
SHRINKAGE = 0.1          # keeps the covariance stable with a limited number of photos
VAL_PERCENTILE = 0.99    # threshold = this percentile of the validation distances

_preprocess = transforms.Compose([
    transforms.Resize(256),
    transforms.CenterCrop(224),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
])


def load_model():
    ckpt = torch.load(MODEL_PATH, map_location="cpu")
    classes = ckpt["classes"]
    model = models.mobilenet_v3_large(weights=None)
    model.classifier[3] = nn.Linear(model.classifier[3].in_features, len(classes))
    model.load_state_dict(ckpt["state_dict"])
    model.eval()
    return model, classes


def list_images(folder):
    if not os.path.isdir(folder):
        return []
    return sorted(os.path.join(folder, f) for f in os.listdir(folder)
                  if f.lower().endswith(IMAGE_EXTS))


@torch.no_grad()
def features_of(model, paths, batch=32):
    """Return an (N, D) tensor of pooled features, skipping unreadable images."""
    out = []
    for i in range(0, len(paths), batch):
        xs = []
        for p in paths[i:i + batch]:
            try:
                with Image.open(p) as im:
                    xs.append(_preprocess(im.convert("RGB")))
            except Exception as e:
                print(f"  skipped {os.path.basename(p)}: {e}")
        if xs:
            f = torch.flatten(model.avgpool(model.features(torch.stack(xs))), 1)
            out.append(f.double())
    return torch.cat(out) if out else torch.empty(0, 0, dtype=torch.double)


def distances(feats, means, precision):
    """Smallest Mahalanobis distance (divided by D) from each feature to any class mean."""
    d = feats.shape[1]
    scores = []
    for f in feats:
        diff = f.unsqueeze(0) - means                     # (C, D)
        d2 = ((diff @ precision) * diff).sum(dim=1)       # (C,)
        scores.append(float(d2.min()) / d)
    return torch.tensor(scores)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--data", default=DEFAULT_DATA, help="folder that contains train/ and val/")
    ap.add_argument("--unknown", default=None, help="optional folder of leaves NOT in the dataset")
    args = ap.parse_args()

    model, classes = load_model()
    print(f"Model has {len(classes)} classes: {', '.join(classes)}")

    train_dir = os.path.join(args.data, "train")
    val_dir = os.path.join(args.data, "val")

    class_feats = []
    for c in classes:
        paths = list_images(os.path.join(train_dir, c))
        if not paths:
            print(f"ERROR: no training photos found in {os.path.join(train_dir, c)}")
            sys.exit(1)
        print(f"Reading {len(paths):4d} training photos of {c} ...")
        class_feats.append(features_of(model, paths))

    means = torch.stack([f.mean(dim=0) for f in class_feats])            # (C, D)
    centered = torch.cat([f - m for f, m in zip(class_feats, means)])    # (N, D)
    n, d = centered.shape
    cov = centered.T @ centered / max(n - 1, 1)
    cov = (1 - SHRINKAGE) * cov + SHRINKAGE * (torch.trace(cov) / d) * torch.eye(d, dtype=torch.double)
    precision = torch.linalg.inv(cov)

    val_scores = []
    for c in classes:
        paths = list_images(os.path.join(val_dir, c))
        if paths:
            val_scores.append(distances(features_of(model, paths), means, precision))
    if not val_scores:
        print(f"ERROR: no validation photos found under {val_dir}")
        sys.exit(1)
    val_scores = torch.cat(val_scores)
    threshold = float(torch.quantile(val_scores, VAL_PERCENTILE))

    print(f"\nValidation photos: {len(val_scores)}")
    print(f"  distance  median {float(val_scores.median()):.3f}   "
          f"95% {float(torch.quantile(val_scores, 0.95)):.3f}   max {float(val_scores.max()):.3f}")
    print(f"  threshold (99% of validation photos pass): {threshold:.3f}")

    if args.unknown:
        paths = list_images(args.unknown)
        if paths:
            u = distances(features_of(model, paths), means, precision)
            rejected = int((u > threshold).sum())
            print(f"\nUnknown leaves in {args.unknown}: {len(u)} photos")
            print(f"  distance  median {float(u.median()):.3f}   min {float(u.min()):.3f}")
            print(f"  rejected at this threshold: {rejected} of {len(u)}")

    torch.save({
        "means": means.float(),
        "precision": precision.float(),
        "threshold": threshold,
        "classes": classes,
    }, STATS_PATH)
    print(f"\nSaved {STATS_PATH}")


if __name__ == "__main__":
    main()

"""
Wraps torchvision.datasets.ImageFolder so training/validation get
consistent transforms, and gives back the standard PyTorch
train/val DataLoaders.

Expected folder layout (ImageFolder infers labels from subfolder names):

    ml/dataset/

from torch.utils.data import DataLoader, random_split
from torchvision import datasets, transforms
import torch

from . import config


def build_transforms(train: bool) -> transforms.Compose:
    if train:
        return transforms.Compose([
            transforms.Resize((config.IMAGE_SIZE, config.IMAGE_SIZE)),
            transforms.RandomHorizontalFlip(),
            transforms.RandomRotation(15),
            transforms.ColorJitter(brightness=0.2, contrast=0.2, saturation=0.2),
            transforms.ToTensor(),
            transforms.Normalize(config.NORMALIZE_MEAN, config.NORMALIZE_STD),
        ])
    return transforms.Compose([
        transforms.Resize((config.IMAGE_SIZE, config.IMAGE_SIZE)),
        transforms.ToTensor(),
        transforms.Normalize(config.NORMALIZE_MEAN, config.NORMALIZE_STD),
    ])


def get_dataloaders():
    if not config.DATASET_DIR.exists():
        raise FileNotFoundError(
            f"Dataset folder not found at {config.DATASET_DIR}. "
            "Create ml/dataset/herbal/ and ml/dataset/non_herbal/ and add images."
        )

    full_dataset = datasets.ImageFolder(str(config.DATASET_DIR))

    if len(full_dataset) == 0:
        raise ValueError(
            f"No images found under {config.DATASET_DIR}. "
            "Add .jpg/.png files to dataset/herbal/ and dataset/non_herbal/."
        )

    # Sanity check: folder names should match what config.py expects,
    # otherwise class indices (0/1) will silently be swapped.
    found_classes = full_dataset.classes
    if sorted(found_classes) != sorted(config.CLASS_NAMES):
        raise ValueError(
            f"Dataset subfolders {found_classes} don't match "
            f"config.CLASS_NAMES {config.CLASS_NAMES}. "
            "Rename the subfolders under ml/dataset/ to match exactly."
        )

    val_size = int(len(full_dataset) * config.VAL_SPLIT)
    train_size = len(full_dataset) - val_size

    generator = torch.Generator().manual_seed(config.RANDOM_SEED)
    train_subset, val_subset = random_split(
        full_dataset, [train_size, val_size], generator=generator
    )

    # ImageFolder applies one transform to the whole dataset, but we want
    # augmentation only on train â€” so wrap each subset with its own transform.
    train_subset.dataset = datasets.ImageFolder(
        str(config.DATASET_DIR), transform=build_transforms(train=True)
    )
    val_subset.dataset = datasets.ImageFolder(
        str(config.DATASET_DIR), transform=build_transforms(train=False)
    )

    train_loader = DataLoader(
        train_subset,
        batch_size=config.BATCH_SIZE,
        shuffle=True,
        num_workers=config.NUM_WORKERS,
    )
    val_loader = DataLoader(
        val_subset,
        batch_size=config.BATCH_SIZE,
        shuffle=False,
        num_workers=config.NUM_WORKERS,
    )

    print(f"Loaded {len(full_dataset)} images "
          f"({train_size} train / {val_size} val). Classes: {full_dataset.classes}")

    return train_loader, val_loader

r"""
Classify one photo with the trained model (leaf_model.pt).

Command line:
    python leaf_predict.py path\to\photo.jpg
    python leaf_predict.py --folder path\to\folder   (test many photos, to tune the limits)

From your own code (for example a web backend):
    from leaf_predict import predict, is_confident
    results = predict("photo.jpg")      # [("Sambong", 0.93), ("Sawansawan", 0.04), ...]
    if not is_confident(results):
        ...  # show NOT_IN_DATASET_MESSAGE

Needs: torch, torchvision, pillow, and leaf_model.pt in the same folder as this file.
For the "unknown leaf" distance check, also run build_leaf_stats.py once to create
leaf_stats.pt (without that file the distance check is simply skipped).
"""

import os
import sys

import torch
import torch.nn as nn
from PIL import Image
from torchvision import models, transforms

_HERE = os.path.dirname(os.path.abspath(__file__))
MODEL_PATH = os.path.join(_HERE, "leaf_model.pt")
STATS_PATH = os.path.join(_HERE, "leaf_stats.pt")

# A result counts as "in the dataset" only if ALL checks pass.
# Tune with:  python leaf_predict.py --folder <photos of leaves NOT in the dataset>
MIN_CONFIDENCE = 0.80   # top guess must be at least this sure (was 0.60)
MIN_MARGIN = 0.30       # top guess must beat the 2nd guess by at least this much
DISTANCE_SCALE = 1.0    # multiplies the distance limit saved in leaf_stats.pt (lower = stricter)

NOT_IN_DATASET_MESSAGE = (
    "This leaf is not in the dataset, or the photo is unclear. "
    "Try again with a clear photo of a single leaf."
)

# Friendly names shown to users. Keys must match the folder names used in training.
COMMON_NAMES = {
    "Blumea_balsamifera": "Sambong",
    "Annona_muricata": "Guyabano",
    "Hyptis_capitata": "Sawansawan",
    "Dracaena_reflexa": "Song of India",
    "Dieffenbachia_seguine": "Dumb cane",
    "Caladium_bicolor": "Caladium",
    "Other_plants": "Other plants",   # only exists in a model trained with the 7th class
}
OTHER_NAME = "Other plants"           # if this is the top guess, the leaf is "not in the dataset"

_preprocess = transforms.Compose([
    transforms.Resize(256),
    transforms.CenterCrop(224),
    transforms.ToTensor(),
    transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
])

_model = None
_classes = None
_stats = None


class Guesses(list):
    """A list of (name, probability) plus the distance to the nearest known plant."""
    distance = None


def _load():
    global _model, _classes, _stats
    if _model is not None:
        return
    ckpt = torch.load(MODEL_PATH, map_location="cpu")
    _classes = ckpt["classes"]
    model = models.mobilenet_v3_large(weights=None)
    model.classifier[3] = nn.Linear(model.classifier[3].in_features, len(_classes))
    model.load_state_dict(ckpt["state_dict"])
    model.eval()
    if os.path.exists(STATS_PATH):
        stats = torch.load(STATS_PATH, map_location="cpu")
        if list(stats["classes"]) == list(_classes):
            _stats = stats
        else:
            print("leaf_stats.pt does not match leaf_model.pt - run build_leaf_stats.py again.")
    else:
        print("leaf_stats.pt not found - skipping the unknown-leaf distance check.")
    _model = model


def _distance(feat):
    """Smallest Mahalanobis distance (divided by D) from one feature vector to any class mean."""
    if _stats is None:
        return None
    diff = feat.unsqueeze(0) - _stats["means"]
    d2 = ((diff @ _stats["precision"]) * diff).sum(dim=1)
    return float(d2.min()) / feat.shape[0]


def predict(image_path, top_k=3):
    """Return the top_k guesses as a list of (name, probability), best first."""
    _load()
    with Image.open(image_path) as im:
        x = _preprocess(im.convert("RGB")).unsqueeze(0)
    with torch.no_grad():
        feats = torch.flatten(_model.avgpool(_model.features(x)), 1)
        probs = _model.classifier(feats).softmax(dim=1)[0]
        dist = _distance(feats[0])
    values, indices = probs.topk(min(top_k, len(_classes)))
    results = Guesses((COMMON_NAMES.get(_classes[i], _classes[i]), float(v))
                      for v, i in zip(values, indices))
    results.distance = dist
    return results


def margin(results):
    """Gap between the best and second-best guess (needs top_k >= 2)."""
    second = results[1][1] if len(results) > 1 else 0.0
    return results[0][1] - second


def too_far(results):
    """True if the leaf looks unlike all six known plants (needs leaf_stats.pt)."""
    d = getattr(results, "distance", None)
    return d is not None and _stats is not None and d > _stats["threshold"] * DISTANCE_SCALE


def is_confident(results):
    """True if the leaf looks like one of the dataset plants, False if it should be 'not found'."""
    if results[0][0] == OTHER_NAME:      # the model says "some other plant"
        return False
    if results[0][1] < MIN_CONFIDENCE or margin(results) < MIN_MARGIN:
        return False
    return not too_far(results)


_IMAGE_EXTS = (".jpg", ".jpeg", ".png", ".webp", ".bmp")


def _run_folder(folder):
    files = sorted(f for f in os.listdir(folder) if f.lower().endswith(_IMAGE_EXTS))
    if not files:
        print("No images found in that folder.")
        return
    print(f"{'file':<32}{'best guess':<16}{'conf':>7}{'margin':>8}{'dist':>8}  verdict")
    accepted = 0
    for f in files:
        try:
            results = predict(os.path.join(folder, f))
        except Exception as e:  # unreadable image
            print(f"{f:<32}error: {e}")
            continue
        ok = is_confident(results)
        accepted += ok
        d = results.distance
        dist = f"{d:8.2f}" if d is not None else f"{'-':>8}"
        print(f"{f[:31]:<32}{results[0][0]:<16}{results[0][1] * 100:6.1f}%{margin(results) * 100:7.1f}%{dist}  "
              f"{'ACCEPTED' if ok else 'not found'}")
    limit = (f"{_stats['threshold'] * DISTANCE_SCALE:.2f}" if _stats is not None else "off")
    print(f"\n{accepted} of {len(files)} accepted "
          f"(MIN_CONFIDENCE={MIN_CONFIDENCE}, MIN_MARGIN={MIN_MARGIN}, distance limit={limit})")


if __name__ == "__main__":
    if len(sys.argv) == 3 and sys.argv[1] == "--folder":
        _run_folder(sys.argv[2])
        sys.exit(0)

        print(f"  distance to nearest known plant: {results.distance:.2f}")
    if not is_confident(results):

"""
A small CNN trained from scratch â€” matches "self-trained CNN" in the
architecture diagram (as opposed to fine-tuning a pretrained backbone).

This is intentionally simple to start: 4 conv blocks + a classifier head.
Once training works end-to-end, swapping in a pretrained backbone
(e.g. torchvision.models.mobilenet_v3_small) is a drop-in change if
accuracy needs a boost later.
"""
import torch.nn as nn

from . import config


class HerbCNN(nn.Module):
    def __init__(self, num_classes: int = len(config.CLASS_NAMES)):
        super().__init__()

        self.features = nn.Sequential(
            self._conv_block(3, 32),
            self._conv_block(32, 64),
            self._conv_block(64, 128),
            self._conv_block(128, 256),
        )

        # config.IMAGE_SIZE gets halved 4 times by the 4 pooling layers above
        reduced_size = config.IMAGE_SIZE // (2 ** 4)
        flattened_dim = 256 * reduced_size * reduced_size

        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Dropout(0.3),
            nn.Linear(flattened_dim, 256),
            nn.ReLU(inplace=True),
            nn.Dropout(0.3),
            nn.Linear(256, num_classes),
        )

    @staticmethod
    def _conv_block(in_channels: int, out_channels: int) -> nn.Sequential:
        return nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2),
        )

    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x

"""
Loads the trained checkpoint and runs inference on a single image.
This is what your FastAPI app should import and call at startup /
per-request, matching the "PyTorch â€” self-trained CNN inference" and
".pt/.pth model file, loaded at startup" boxes in the architecture diagram.

Example (inside a FastAPI route):

    from .ml.predict import HerbPredictor

    predictor = HerbPredictor()  # load once, at app startup

    @app.post("/classify")
    async def classify(file: UploadFile):
        image_bytes = await file.read()
        label, confidence = predictor.predict(image_bytes)
        return {"label": label, "confidence": confidence}
"""
import io

import torch
from PIL import Image
from torchvision import transforms

from . import config
from .model import HerbCNN


class HerbPredictor:
    def __init__(self, checkpoint_path=None):
        checkpoint_path = checkpoint_path or config.MODEL_OUT_PATH
        if not checkpoint_path.exists():
            raise FileNotFoundError(
                f"No trained model found at {checkpoint_path}. "
                "Run `python -m ml.train` first."
            )

        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        checkpoint = torch.load(checkpoint_path, map_location=self.device)

        self.class_names = checkpoint["class_names"]
        image_size = checkpoint["image_size"]

        self.model = HerbCNN(num_classes=len(self.class_names))
        self.model.load_state_dict(checkpoint["model_state_dict"])
        self.model.to(self.device)
        self.model.eval()

        self.transform = transforms.Compose([
            transforms.Resize((image_size, image_size)),
            transforms.ToTensor(),
            transforms.Normalize(config.NORMALIZE_MEAN, config.NORMALIZE_STD),
        ])

        print(f"HerbPredictor loaded on {self.device} "
              f"(val_acc at save time: {checkpoint.get('val_acc', 'unknown')})")

    def predict(self, image_bytes: bytes) -> tuple[str, float]:
        """Returns (predicted_label, confidence_percent)."""
        image = Image.open(io.BytesIO(image_bytes)).convert("RGB")
        tensor = self.transform(image).unsqueeze(0).to(self.device)

        with torch.no_grad():
            outputs = self.model(tensor)
            probs = torch.softmax(outputs, dim=1)[0]
            confidence, predicted_idx = torch.max(probs, dim=0)

        label = self.class_names[predicted_idx.item()]
        return label, round(confidence.item() * 100, 1)

"""
Entry point for training the herbal/non-herbal classifier.

Usage (from the backend/app directory):
    python -m ml.train

Produces: ml/checkpoints/herb_classifier.pt
This is the file your FastAPI startup code (per the architecture diagram,
"loaded at startup") should load for inference.
"""
import torch
import torch.nn as nn
from torch.optim import Adam

from . import config
from .dataset import get_dataloaders
from .model import HerbCNN


def train():
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Using device: {device}")

    train_loader, val_loader = get_dataloaders()

    model = HerbCNN().to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = Adam(model.parameters(), lr=config.LEARNING_RATE)

    config.CHECKPOINT_DIR.mkdir(parents=True, exist_ok=True)
    best_val_acc = 0.0

    for epoch in range(1, config.NUM_EPOCHS + 1):
        model.train()
        running_loss = 0.0
        for images, labels in train_loader:
            images, labels = images.to(device), labels.to(device)

            optimizer.zero_grad()
            outputs = model(images)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

            running_loss += loss.item() * images.size(0)

        train_loss = running_loss / len(train_loader.dataset)
        val_loss, val_acc = _evaluate(model, val_loader, criterion, device)

        print(f"Epoch {epoch:>2}/{config.NUM_EPOCHS} | "
              f"train_loss={train_loss:.4f} | "
              f"val_loss={val_loss:.4f} | val_acc={val_acc:.2%}")

        if val_acc > best_val_acc:
            best_val_acc = val_acc
            torch.save(
                {
                    "model_state_dict": model.state_dict(),
                    "class_names": config.CLASS_NAMES,
                    "image_size": config.IMAGE_SIZE,
                    "val_acc": val_acc,
                },
                config.MODEL_OUT_PATH,
            )
            print(f"  -> New best model saved (val_acc={val_acc:.2%})")

    print(f"\nTraining complete. Best val_acc={best_val_acc:.2%}")
    print(f"Model saved to: {config.MODEL_OUT_PATH}")


def _evaluate(model, val_loader, criterion, device):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0

    with torch.no_grad():
        for images, labels in val_loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)

            running_loss += loss.item() * images.size(0)
            predicted = outputs.argmax(dim=1)
            correct += (predicted == labels).sum().item()
            total += labels.size(0)

    return running_loss / total, correct / total


if __name__ == "__main__":
    train()

from ..schemas import UserRegister, UserLogin, TokenResponse

router = APIRouter(prefix="/auth", tags=["auth"])


@router.post("/register", response_model=TokenResponse, status_code=status.HTTP_201_CREATED)
def register(payload: UserRegister, db: Session = Depends(get_db)):
    existing = db.query(User).filter(User.email == payload.email).first()
    if existing:
        raise HTTPException(status_code=400, detail="Email already registered")

    new_user = User(
        display_name=payload.display_name,
        email=payload.email,
        password_hash=hash_password(payload.password),
    )
    db.add(new_user)
    db.commit()
    db.refresh(new_user)

    token = create_access_token({"sub": str(new_user.id)})
    return TokenResponse(access_token=token, user=new_user)


@router.post("/login", response_model=TokenResponse)
def login(payload: UserLogin, db: Session = Depends(get_db)):
    user = db.query(User).filter(User.email == payload.email).first()
    if not user or not verify_password(payload.password, user.password_hash):
        raise HTTPException(status_code=401, detail="Invalid email or password")

    token = create_access_token({"sub": str(user.id)})
    return TokenResponse(access_token=token, user=user)

# Scientific names of the herbal plants. Everything else is non-herbal.
HERBAL_SPECIES = [
    "blumea balsamifera",   # Sambong
    "annona muricata",      # Guyabano
    "hyptis capitata",      # Sawansawan
]


def _category(name: str) -> str:
    lowered = (name or "").lower()
    return "Herbal" if any(s in lowered for s in HERBAL_SPECIES) else "Non-herbal"


def _to_dict(item: LeafItem) -> dict:
    return {
        "id": item.id,
        "name": item.name,
        "category": _category(item.name),
        "description": item.description,
        "benefits": item.benefits,
        "traditional_uses": item.traditional_uses,
        "risks": item.risks,
        "precautions": item.precautions,
    }


@router.get("/")
def list_herbs(db: Session = Depends(get_db)):
    items = db.query(LeafItem).order_by(LeafItem.id).all()
    return [_to_dict(i) for i in items]


@router.get("/lookup")
def lookup_herb(name: str, db: Session = Depends(get_db)):
    # The stored name looks like "Sambong (Alib-on, Gabon) - Blumea balsamifera",
    # so a match on the scientific name (or any common name) finds the row.
    item = db.query(LeafItem).filter(LeafItem.name.ilike(f"%{name}%")).first()
    if item is None:
        raise HTTPException(status_code=404, detail="Herb not found")
    return _to_dict(item)
